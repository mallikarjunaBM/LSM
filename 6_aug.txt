import machine

import math
import struct

class KalmanFilter:
    def __init__(self, q=0.01, r=0.1):
        self.q = q
        self.r = r
        self.x = 0.0
        self.p = 1.0
        self.k = 0.0

    def update(self, measurement):
        self.p = self.p + self.q
        self.k = self.p / (self.p + self.r)
        self.x = self.x + self.k * (measurement - self.x)
        self.p = (1 - self.k) * self.p
        return self.x

def read_lsm303():
    data_a = i2c.readfrom_mem(0x19, 0x28 | 0x80, 6)
    ax, ay, az = struct.unpack('<hhh', data_a)
    
    data_m = i2c.readfrom_mem(0x1E, 0x03, 6)
    mx, mz, my = struct.unpack('>hhh', data_m)
    
    return ax, ay, az, mx, my, mz

def read_icp10111():
    try:
        i2c.writeto(0x63, b'\x68\x25')
        time.sleep_ms(2)
        data = i2c.readfrom(0x63, 9)
        
        t_raw = (data[0] << 8) | data[1]
        p_raw = (data[3] << 16) | (data[4] << 8) | data[6]
        
        temperature = -45.0 + (175.0 * t_raw / 65536.0)
        pressure = p_raw / 100.0
        return temperature, pressure
    except OSError:
        return 0.0, 0.0
